# 开发环境依赖包
# MatuX Backend 开发工具链
#
# 版本策略：全部使用精确版本（==）锁定，避免 CI 镜像随上游发布而漂移，
# 导致「本地绿 / CI 红」或反之。升级时改本文件并单独提交。

# 代码格式化工具
black==26.10.1
isort==9.0.2

# 代码质量检查工具
# 说明：flake8 插件只保留与根 .flake8 的 select=E,W,F,C,B 对应的三个：
#   - flake8-bugbear      -> B 码
#   - flake8-comprehensions-> C4xx 码
#   - flake8 本身         -> E/W/F 码与 mccabe 的 C901
# 已移除（其码段不在 select 内，且与 black/isort 冲突）：
#   - flake8-quotes      -> Q 码（默认单引号，与 black 双引号冲突）
#   - flake8-docstrings  -> D 码
#   - flake8-import-order-> I 码（与 isort 排序规则打架）
#   - flake8-print       -> T20 码
# 机械类问题（F401/W293/E712 等）改由 ruff 批量处理。
flake8==7.5.0
flake8-bugbear==26.10.27
flake8-comprehensions==3.17.0
ruff==0.17.3

# 类型检查工具
mypy==2.4.1

# 测试工具
pytest==9.2.0
pytest-cov==7.1.0
pytest-asyncio==1.4.1.post1
pytest-mock==3.17.0
# pytest 的 --timeout=30 依赖本插件；缺失时 pytest 直接 UsageError 退出，
# 0 个用例被执行（Issue #32 的 CI 假绿根因）。
pytest-timeout==2.4.0
